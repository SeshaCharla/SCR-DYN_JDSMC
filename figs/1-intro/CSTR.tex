% --- Tank body ---

    (-\tw, 0) -- (-\tw, \th) arc(180:360:{\tw} and {\ey}) -- (\tw, 0) arc(0:-180:{\tw} and {\ey});

% --- Catalyst surface (hatched strip inside, near bottom) ---

\node[font=\footnotesize, gray!50!black, anchor=south] at (0, 1.0) {catalyst surface $(\Gamma)$};

% --- Inlet (horizontal arrow into left wall, upper region) ---
\draw[->, line width=2pt] (-6.2, 4.5) -- (-\tw-0.05, 4.5);
\node[lbl, anchor=south west] at (-6.2, 4.6) {$F,\; T,\; [\text{NO}_x]^{in},\; [\text{NH}_3]^{in}$};

% --- Outlet (horizontal arrow from right wall, lower region) ---
\draw[->, line width=2pt] (\tw+0.05, 1.7) -- (6.2, 1.7);
